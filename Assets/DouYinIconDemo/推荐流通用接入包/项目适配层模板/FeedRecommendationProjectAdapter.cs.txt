public static class MyGameFeedAdapter
{
    private static bool registered;
    private static FeedVideoPlayerController video;

    public static void Register()
    {
        if (registered) return;
        registered = true;

        FeedRecommendationManager manager = FeedRecommendationManager.Instance;
        manager.SchemePrepared += OnPrepared;
        manager.CardShown += OnCardShown;
        manager.CardHidden += OnCardHidden;
        manager.FeedEntered += OnFeedEntered;
        manager.SchemeEntered += OnFirstEntered;
        manager.FeedExited += OnFeedExited;
        manager.ConfigurationFailed += OnConfigurationFailed;
        manager.FullGameHidden += OnFullGameHidden;
        manager.FullGameShown += OnFullGameShown;
        manager.SessionEnded += OnSessionEnded;
    }

    {
        CloseVideo();
        // 这里只加载场景、关卡、UI、输入、AudioClip 和视频首帧。
        // 禁止启动 AI、配音、字幕、倒计时、存档和奖励。
        switch (context.Scheme)
        {
            case FeedExperienceScheme.Video:
                PrepareVideo(context.Payload);
                break;
            case FeedExperienceScheme.TrialLevel:
                // TODO：预加载 context.Payload 指定的试玩关卡。
                // 可玩画面渲染完成后调用 ReportSceneReady()。
                break;
            case FeedExperienceScheme.NewPlayerGuide:
                // TODO：预加载 context.Payload 指定的新手引导。
                // 可玩画面渲染完成后调用 ReportSceneReady()。
                break;
        }
    }

    private static void OnCardShown(FeedLaunchContext context)
    {
        if (video != null) video.SetCardVisible(true);

        // TODO：从这里开始或恢复卡片 AI、动画和试玩计时。
        // 配音 Play、字幕时钟清零和首条字幕必须在这个方法的同一帧放行。
        // 不要从 SchemePrepared、资源加载完成、ReportScene 或 FeedEnter 启动卡片配音。
    }

    private static void OnCardHidden(FeedLaunchContext context)
    {
        if (video != null) video.SetCardVisible(false);
        // TODO：暂停卡片 AI、配音、字幕和试玩计时，保留当前位置供下次 OnShow 恢复。
    }

    private static void OnFeedEntered(FeedLaunchContext context)
    {
        // 每次从推荐流进入完整小游戏都会触发；用于恢复完整游戏。
        // TODO：恢复目标游戏自己的暂停系统，不要从头重播卡片配音。
    }

    private static void OnFirstEntered(FeedLaunchContext context)
    {
        // 整个推荐流会话只触发一次；用于登录、active、正式存档和完整游戏承接。
        if (video != null)
        {
            video.Close();
            video = null;
        }
        // 编辑器预览不登录、不写正式存档、不发业务奖励。
        if (FeedRecommendationManager.Instance.IsEditorPreview) return;
        // TODO：执行目标游戏第一次正式进入逻辑。
    }

    private static void OnFeedExited(FeedLaunchContext context)
    {
        // TODO：暂停完整小游戏；等待后续真实 OnShow 再恢复卡片体验。
    }

    private static void OnFullGameHidden(FeedLaunchContext context)
    {
        // TODO：用项目暂停系统暂停完整游戏的声音、输入和计时。
    }

    private static void OnFullGameShown(FeedLaunchContext context)
    {
        // TODO：恢复完整游戏，不重做登录、初始化或首次进入奖励。
    }

    private static void CloseVideo()
    {
        if (video == null) return;
        FeedVideoPlayerController previous = video;
        video = null;
        previous.Close();
    }

    private static void OnSessionEnded(FeedLaunchContext context)
    {
        CloseVideo();
        // TODO：取消本会话的异步加载、配音、字幕和临时暂停占用。
    }

    private static void PrepareVideo(string url)
    {
        CloseVideo();
        FeedRecommendationManager manager = FeedRecommendationManager.Instance;
        int version = manager.LaunchVersion;
        FeedVideoPlayerController current = FeedVideoPlayerController.Create();
        video = current;
        current.Ready += () =>
        {
            if (manager.IsCurrentSession(version) && video == current) manager.ReportSceneReady();
        };
        current.Failed += message =>
        {
            if (!manager.IsCurrentSession(version) || video != current) return;
            Debug.LogWarning("[推荐流适配] 视频失败，加载安全兜底页面：" + message);
            CloseVideo();
            // TODO：加载首页，完成后仅在 manager.IsCurrentSession(version) 时上报就绪。
        };
        current.PrepareAndPlay(url);
    }

    }
